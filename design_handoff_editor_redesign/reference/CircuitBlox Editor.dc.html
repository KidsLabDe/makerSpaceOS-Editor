import board, neopixel, time
from digitalio import DigitalInOut, Direction, Pull

pixels = neopixel.NeoPixel(board.GP0, 8)
button = DigitalInOut(board.GP1)
button.switch_to_input(pull=Pull.UP)

# Hauptschleife
while True:
    pixels.fill((255, 0, 0))
    time.sleep(0.5)
    if not button.value:
        print("Knopf gedrückt!")
        time.sleep(0.2)
